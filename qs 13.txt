Q13. Kruskal's Algorithm Using Adjacency List

Language: Python

Code:
def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, a, b):
    a = find(parent, a)
    b = find(parent, b)
    if a != b:
        parent[b] = a
        return True
    return False

n = int(input("Enter number of vertices: "))
e = int(input("Enter number of edges: "))
graph = [[] for i in range(n)]
edges = []

print("Enter edges: u v weight")
for i in range(e):
    u, v, w = map(int, input().split())
    graph[u].append((v, w))
    graph[v].append((u, w))
    edges.append((w, u, v))

edges.sort()
parent = list(range(n))
total = 0

print("MST Edges:")
for w, u, v in edges:
    if union(parent, u, v):
        print(u, "-", v, ":", w)
        total += w

print("Total Weight:", total)

Sample Input:
4
5
0 1 10
0 2 6
0 3 5
1 3 15
2 3 4

Sample Output:
MST Edges:
2 - 3 : 4
0 - 3 : 5
0 - 1 : 10
Total Weight: 19

Vertices are numbered 0 to n-1. A connected graph is required for a complete MST.
