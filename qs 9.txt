Q9. BST Insertion and Tree Traversals

Language: Python

Code:
class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def insert(root, data):
    if root is None:
        return Node(data)
    if data < root.data:
        root.left = insert(root.left, data)
    elif data > root.data:
        root.right = insert(root.right, data)
    return root

def inorder(root):
    if root:
        inorder(root.left)
        print(root.data, end=" ")
        inorder(root.right)

def preorder(root):
    if root:
        print(root.data, end=" ")
        preorder(root.left)
        preorder(root.right)

def postorder(root):
    if root:
        postorder(root.left)
        postorder(root.right)
        print(root.data, end=" ")

root = None
n = int(input("Enter number of nodes: "))
for i in range(n):
    x = int(input())
    root = insert(root, x)

print("Inorder:")
inorder(root)
print("\nPreorder:")
preorder(root)
print("\nPostorder:")
postorder(root)

Sample Input:
5
50
30
70
20
40

Sample Output:
Inorder:
20 30 40 50 70
Preorder:
50 30 20 40 70
Postorder:
20 40 30 70 50

Duplicate BST values are ignored.
