Q15. Heap Sort

Language: Python

Code:
def heapify(a, n, i):
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2
    if left < n and a[left] > a[largest]:
        largest = left
    if right < n and a[right] > a[largest]:
        largest = right
    if largest != i:
        a[i], a[largest] = a[largest], a[i]
        heapify(a, n, largest)

def heap_sort(a):
    n = len(a)
    # Build max heap
    for i in range(n // 2 - 1, -1, -1):
        heapify(a, n, i)
    # Move maximum to the end
    for i in range(n - 1, 0, -1):
        a[0], a[i] = a[i], a[0]
        heapify(a, i, 0)

a = list(map(int, input("Enter array: ").split()))
heap_sort(a)
print("Sorted Array:", a)

Sample Input:
5 3 8 1 2

Sample Output:
Sorted Array: [1, 2, 3, 5, 8]
