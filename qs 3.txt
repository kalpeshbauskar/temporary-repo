3.	Store student ID, name, and CGPA. Implement Bubble Sort or Selection Sort to arrange records by CGPA in ascending order.
Language: C++

Code:
#include <iostream>
using namespace std;

struct Student {
    int id;
    string name;
    float cgpa;
};

int main() {
    int n;
    Student s[100], temp;

    cout << "Enter number of students: ";
    cin >> n;

    for (int i = 0; i < n; i++) {
        cin >> s[i].id >> s[i].name >> s[i].cgpa;
    }

    // Selection Sort by CGPA
    for (int i = 0; i < n - 1; i++) {
        int min = i;
        for (int j = i + 1; j < n; j++) {
            if (s[j].cgpa < s[min].cgpa)
                min = j;
        }
        temp = s[i];
        s[i] = s[min];
        s[min] = temp;
    }

    cout << "Sorted Students:\n";
    for (int i = 0; i < n; i++) {
        cout << s[i].id << " "
             << s[i].name << " "
             << s[i].cgpa << endl;
    }

    return 0;
}

Sample Input:
3
101 Rahul 8.5
102 Amit 9.2
103 Raj 7.8

Sample Output:
Sorted Students:
103 Raj 7.8
101 Rahul 8.5
102 Amit 9.2
