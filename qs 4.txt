4.	Store student IDs in an array, sort them in ascending order, and implement Binary Search to find a given student ID.

Language: C++

// Q4 is identical to Q2 in the supplied question bank.
// Use the complete Q2 program without changes.

Code:
#include <iostream>
using namespace std;

int main() {
    int n, a[100], key;

    cout << "Enter number of students: ";
    cin >> n;

    for (int i = 0; i < n; i++)
        cin >> a[i];

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (a[j] > a[j + 1])
                swap(a[j], a[j + 1]);
        }
    }

    cout << "Enter ID to search: ";
    cin >> key;

    int low = 0, high = n - 1;
    bool found = false;

    while (low <= high) {
        int mid = (low + high) / 2;

        if (a[mid] == key) {
            found = true;
            break;
        }
        else if (a[mid] < key)
            low = mid + 1;
        else
            high = mid - 1;
    }

    if (found)
        cout << "ID Found";
    else
        cout << "ID Not Found";

    return 0;
}

Sample Input:
5
50 20 40 10 30
40

Sample Output:
ID Found
